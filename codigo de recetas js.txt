function obtenerContextoAudio() {
    if (!audioCtxGlobal) {
        audioCtxGlobal = new (window.AudioContext || window.webkitAudioContext)();
    }
    if (audioCtxGlobal.state === 'suspended') {
        audioCtxGlobal.resume();
    }
    return audioCtxGlobal;
}

function verificarPIN() {
    var pinIngresado = document.getElementById('input-pin').value;
    if (pinIngresado === PIN_NORMAL) {
        nivelUsuario = 1;
        actualizarEstilosTarjetas();
        irPantalla('screen-inicio');
    } else if (pinIngresado === PIN_VIP) {
        nivelUsuario = 2;
        actualizarEstilosTarjetas();
        irPantalla('screen-inicio');
    } else if (pinIngresado === PIN_PREMIUM || pinIngresado === PIN_MAESTRO) {
        nivelUsuario = 3;
        actualizarEstilosTarjetas();
        irPantalla('screen-inicio');
    } else {
        var msg = idiomaActual === 'es' ? "El PIN ingresado es incorrecto. Intenta de nuevo." : (idiomaActual === 'fr' ? "Le code PIN est incorrect. Réessayez." : "The entered PIN is incorrect. Try again.");
        var title = idiomaActual === 'es' ? "Acceso Denegado" : (idiomaActual === 'fr' ? "Accès Refusé" : "Access Denied");
        mostrarAlerta(msg, title);
    }
}

function actualizarEstilosTarjetas() {
    var recetasVIP = ['card-coctel', 'card-bandeja', 'card-arroz'];
    recetasVIP.forEach(function(id) {
        var card = document.getElementById(id);
        if (card) {
            if (nivelUsuario >= 2) {
                card.classList.remove('locked-card');
                var badge = card.querySelector('.badge-lock');
                if (badge) { 
                    badge.innerText = idiomaActual === 'es' ? "⭐ Desbloqueada" : (idiomaActual === 'fr' ? "⭐ Débloquée" : "⭐ Unlocked"); 
                    badge.style.background = "#EFECE4"; 
                    badge.style.color = "#436143"; 
                }
            } else {
                card.classList.add('locked-card');
            }
        }
    });

    var recetasPremium = ['card-ajiaco', 'card-paparellena', 'card-postretresleches'];
    recetasPremium.forEach(function(id) {
        var card = document.getElementById(id);
        if (card) {
            if (nivelUsuario >= 3) {
                card.classList.remove('locked-card');
                var badge = card.querySelector('.badge-lock');
                if (badge) { 
                    badge.innerText = idiomaActual === 'es' ? "💎 Desbloqueada" : (idiomaActual === 'fr' ? "💎 Débloquée" : "💎 Unlocked"); 
                    badge.style.background = "#EFECE4"; 
                    badge.style.color = "#436143"; 
                }
            } else {
                card.classList.add('locked-card');
            }
        }
    });
}

function mostrarAlerta(mensaje, titulo) {
    document.getElementById('modal-titulo').innerText = titulo || (idiomaActual === 'es' ? "¡Un momento!" : (idiomaActual === 'fr' ? "Un moment !" : "Just a moment!"));
    document.getElementById('modal-mensaje').innerText = mensaje;
    document.getElementById('custom-modal').style.display = 'flex';
}

function cerrarAlerta(event) {
    if (event) { event.preventDefault(); event.stopPropagation(); }
    document.getElementById('custom-modal').style.display = 'none';
}

function cerrarUpsell(event) {
    if (event) { event.preventDefault(); event.stopPropagation(); }
    document.getElementById('custom-upsell').style.display = 'none';
    var inp = document.getElementById('input-pin-upgrade');
    if (inp) inp.value = "";
}

function intentarUpgrade(event) {
    if (event) { event.preventDefault(); event.stopPropagation(); }
    var pinVal = document.getElementById('input-pin-upgrade').value;
    
    if (nivelRequeridoEnModal === 'premium') {

            nivelUsuario = 3;
            actualizarEstilosTarjetas();
            cerrarUpsell();
            mostrarAlerta(
                idiomaActual === 'es' ? "¡Has desbloqueado el Paquete Premium Completo (9 Recetas)!" : (idiomaActual === 'fr' ? "Vous avez débloqué le Pack Premium Complet !" : "You unlocked Full Premium Package (9 Recipes)!"),

            );
        } else {
            cerrarUpsell();
            var alertMsg = idiomaActual === 'es' ? "PIN no válido para el nivel Premium. Verifica tu clave." : "Invalid PIN for Premium level. Please check your key.";
            var alertTitle = idiomaActual === 'es' ? "PIN Incorrecto" : "Incorrect PIN";
            mostrarAlerta(alertMsg, alertTitle);
        }
    } 
    else if (nivelRequeridoEnModal === 'vip') {
        if (pinVal === PIN_VIP || pinVal === PIN_PREMIUM || pinVal === PIN_MAESTRO) {
            nivelUsuario = (pinVal === PIN_PREMIUM || pinVal === PIN_MAESTRO) ? 3 : 2;
            actualizarEstilosTarjetas();
            cerrarUpsell();
            mostrarAlerta(
                nivelUsuario === 3 
                ? (idiomaActual === 'es' ? "¡Has desbloqueado el Paquete Premium Completo!" : "You unlocked Premium Package!")
                : (idiomaActual === 'es' ? "¡Has desbloqueado el Paquete VIP (6 Recetas)!" : "You unlocked VIP Package (6 Recipes)!"),
                idiomaActual === 'es' ? "¡Recetas Desbloqueadas!" : "Recipes Unlocked!"
            );
        } else {
            cerrarUpsell();
            var alertMsgVIP = idiomaActual === 'es' ? "PIN no válido para el nivel VIP. Verifica tu clave." : "Invalid PIN for VIP level. Please check your key.";
            var alertTitleVIP = idiomaActual === 'es' ? "PIN Incorrecto" : "Incorrect PIN";
            mostrarAlerta(alertMsgVIP, alertTitleVIP);
        }
    }
}

function cambiarIdioma(lang) {
    idiomaActual = lang;
    document.querySelectorAll('.lang-btn').forEach(function(btn) { btn.classList.remove('active'); });
    if (lang === 'es') document.getElementById('btn-lang-es').classList.add('active');
    if (lang === 'en') document.getElementById('btn-lang-en').classList.add('active');
    if (lang === 'fr') document.getElementById('btn-lang-fr').classList.add('active');

    traducirInterfaz();

    if (document.getElementById('screen-receta').classList.contains('active')) {
        if (preparandoAji) { irAIngredientesAji(); } 
        else if (preparandoPatacon) { irAIngredientesPatacon(); }
        else if (preparandoChimichurri) { irAIngredientesChimichurri(); }
        else { generarReceta(); }
    } else if (document.getElementById('screen-pasos').classList.contains('active')) {
        renderizarPaso();
    }
}

function getProteinaNombre() {
    var protLC = proteinaElegida.toLowerCase();
    if (protLC.includes('pollo')) return idiomaActual === 'es' ? "pollo desmechado" : (idiomaActual === 'fr' ? "poulet effiloché" : "shredded chicken");
    if (protLC.includes('carne') || protLC.includes('res')) {
        if (recetaElegida === "Empanadas Crujientes" || recetaElegida === "Papa Rellena") {
            return idiomaActual === 'es' ? "carne de res molida" : (idiomaActual === 'fr' ? "bœuf haché" : "ground beef");
        } else {
            return idiomaActual === 'es' ? "carne de res" : (idiomaActual === 'fr' ? "bœuf" : "beef");
        }
    }
    if (protLC.includes('cerdo')) return idiomaActual === 'es' ? "cerdo" : (idiomaActual === 'fr' ? "porc" : "pork");
    if (protLC.includes('pescado')) return idiomaActual === 'es' ? "pescado" : (idiomaActual === 'fr' ? "poisson" : "fish");
    if (protLC.includes('vegetales')) return idiomaActual === 'es' ? "vegetales extra" : (idiomaActual === 'fr' ? "légumes extra" : "extra veggies");
    return "";
}

function convertirAFraccion(decimal) {
    if (decimal === 0) return "";
    if (decimal < 0.15) return "1/8";
    if (decimal < 0.3) return "1/4";
    if (decimal < 0.45) return "1/3";
    if (decimal < 0.65) return "1/2";
    if (decimal < 0.85) return "3/4";
    return ""; 
}

function formatearCantidad(cantidadBase, porciones, idIngrediente, unidadEs, unidadEn) {
    var cantTotal = cantidadBase * porciones;
    var unidad = idiomaActual === 'es' ? unidadEs : unidadEn;

    if (unidad === "litros" || unidad === "liters") {
        if (cantTotal < 1) {
            return { cant: Math.round(cantTotal * 1000), uni: "ml" };
        }
    }

    var entera = Math.floor(cantTotal);
    var decimal = cantTotal - entera;

    if (decimal >= 0.85) {
        entera += 1;
        decimal = 0;
    }

    var fraccion = convertirAFraccion(decimal);
    var cantFinal = "";

    if (entera > 0 && fraccion !== "") cantFinal = entera + " " + fraccion;
    else if (entera > 0) cantFinal = entera.toString();
    else if (fraccion !== "") cantFinal = fraccion;
    else cantFinal = "1";

    if (entera === 0 || (entera === 1 && decimal === 0)) {
        if (idiomaActual === 'es') {
            if (unidad === "unidades") unidad = "unidad";
            if (unidad === "tallos") unidad = "tallo";
            if (unidad === "tomates") unidad = "tomate";
            if (unidad === "manojos") unidad = "manojo";
            if (unidad === "cdtas") unidad = "cdta";
            if (unidad === "cdas") unidad = "cda";
            if (unidad === "pizcas") unidad = "pizca";
            if (unidad === "paquetes") unidad = "paquete";
            if (unidad === "hojas") unidad = "hoja";
            if (unidad === "ramitas") unidad = "ramita";
            if (unidad === "dientes") unidad = "diente";
        } else {
            if (unidad === "units") unidad = "unit";
            if (unidad === "stalks") unidad = "stalk";
            if (unidad === "tomatoes") unidad = "tomato";
            if (unidad === "bunches") unidad = "bunch";
            if (unidad === "pinches") unidad = "pinch";
            if (unidad === "packs") unidad = "pack";
            if (unidad === "leaves") unidad = "leaf";
            if (unidad === "sprigs") unidad = "sprig";
            if (unidad === "cloves") unidad = "clove";
        }
    }

    return { cant: cantFinal, uni: unidad };
}

function seleccionarPorciones(cantidad, elementoBtn, event) {
    if (event) event.stopPropagation();
    if (elementoBtn.classList.contains('selected')) {
        porcionesElegidas = 0;
        elementoBtn.classList.remove('selected');
    } else {
        porcionesElegidas = Math.max(0, cantidad);
        document.querySelectorAll('.opcion-btn').forEach(function(btn) { btn.classList.remove('selected'); });
        elementoBtn.classList.add('selected');
        var btnMas = document.getElementById('btn-mas');
        if (btnMas) btnMas.innerText = '+';
    }
    actualizarPreviewPersonas(porcionesElegidas);
}

function seleccionarProteina(nombreProteina, elementoCard) {
    if (elementoCard.classList.contains('selected')) {
        proteinaElegida = "";
        elementoCard.classList.remove('selected');
    } else {
        proteinaElegida = nombreProteina;
        document.querySelectorAll('.proteina-card').forEach(function(card) { card.classList.remove('selected'); });
        elementoCard.classList.add('selected');
    }
}

/* ACTUALIZADOR DE LA BARRA DE PROGRESO DE FONDO EN EL TEMPORIZADOR */
function actualizarBarraProgresoFondo() {
    var box = document.getElementById('timer-box');
    if (!box) return;

        box.style.backgroundImage = 'none';

    }

/* DETECCIÓN E INYECCIÓN DEL TEMPORIZADOR CRONÓMETRO */
function evaluarYTemporizadorPaso(textoPaso) {
    detenerTemporizador();
    timerTerminado = false;
    var contenedor = document.getElementById('contenedor-temporizador-paso');
    if (!contenedor) return;
    contenedor.innerHTML = "";

    var regex = /(\d+)\s*(?:a|-)?\s*(\d+)?\s*(?:minutos|min|m\b)/i;
    var coincidencia = textoPaso.match(regex);

    if (coincidencia) {
        var minutos = coincidencia[2] ? parseInt(coincidencia[2]) : parseInt(coincidencia[1]);
        if (!isNaN(minutos) && minutos > 0) {
            timerTiempoSugeridoInicial = minutos * 60;
            timerTiempoTotal = timerTiempoSugeridoInicial;
            timerTiempoRestante = timerTiempoTotal;
            
            var htmlTimer = `
                <div class="timer-container" id="timer-box" onclick="gestionarTapTemporizador(event)">
                    <div class="timer-click-area" onclick="abrirPickerTiempo()" title="Toca para ajustar tiempo">
                        <span class="timer-clock-icon" id="timer-icon">⏱️</span>

                            <span style="font-size: 10px; font-weight: 700; text-transform: uppercase; color: #5B6B5B; display: block;">${idiomaActual === 'es' ? 'Temporizador (Toca)' : (idiomaActual === 'fr' ? 'Minuteur (Toucher)' : 'Timer (Tap)')}</span>

                        </div>
                    </div>
                    <div class="timer-controls">
                        <button type="button" class="btn-timer" id="btn-start-timer" onclick="toggleTemporizador()">${idiomaActual === 'es' ? 'Iniciar' : (idiomaActual === 'fr' ? 'Démarrer' : 'Start')}</button>

function habilitarArrastreWheel(el) {
    let isDown = false;
    let startY;
    let scrollTop;

    el.addEventListener('mousedown', (e) => {
        isDown = true;
        startY = e.pageY - el.offsetTop;
        scrollTop = el.scrollTop;
    });

    el.addEventListener('mouseleave', () => { isDown = false; });
    el.addEventListener('mouseup', () => { isDown = false; });

    el.addEventListener('mousemove', (e) => {
        if(!isDown) return;
        e.preventDefault();
        const y = e.pageY - el.offsetTop;
        const walk = (y - startY) * 1.5;
        el.scrollTop = scrollTop - walk;
    });
}

function construirRuedaValores(idContenedor, maxValor) {
    var el = document.getElementById(idContenedor);
    if (!el) return;
    el.innerHTML = '<div class="wheel-spacer"></div>';
    
    for (var i = 0; i <= maxValor; i++) {
        var valTxt = i < 10 ? '0' + i : '' + i;
        var item = document.createElement('div');
        item.className = 'wheel-item';
        item.innerText = valTxt;
        item.dataset.val = i;
        el.appendChild(item);
    }
    el.innerHTML += '<div class="wheel-spacer"></div>';

    el.onscroll = function() {
        var index = Math.round(el.scrollTop / 40);
        var items = el.querySelectorAll('.wheel-item');
        items.forEach(function(it, idx) {
            if (idx === index) it.classList.add('selected');
            else it.classList.remove('selected');
        });
    };

    habilitarArrastreWheel(el);
}

    var segActual = timerTiempoRestante % 60;

/* FUNCIÓN DE FLECHAS TOTALMENTE CORREGIDA */
function ajustarRuedaPaso(idContenedor, direccion) {
    if (idContenedor === 'minutos') idContenedor = 'wheel-minutos';
    if (idContenedor === 'segundos') idContenedor = 'wheel-segundos';

    const contenedor = document.getElementById(idContenedor);
    if (!contenedor) return;

    const alturaItem = 40;
    const posicionActual = contenedor.scrollTop;
    const itemActual = Math.round(posicionActual / alturaItem);
    const nuevoItem = itemActual + direccion;

    if (nuevoItem < 0) return;

    contenedor.scrollTop = nuevoItem * alturaItem;
    contenedor.dispatchEvent(new Event('scroll'));
}

function guardarTiempoPicker() {
    var wMin = document.getElementById('wheel-minutos');
    var wSeg = document.getElementById('wheel-segundos');

    var idxMin = Math.round(wMin.scrollTop / 40);
    var idxSeg = Math.round(wSeg.scrollTop / 40);

    var totalSeg = (idxMin * 60) + idxSeg;

    if (totalSeg > 0) {
        timerTiempoTotal = totalSeg;
        timerTiempoRestante = totalSeg;

        var display = document.getElementById('timer-display');
        if (display) display.innerText = formatearTiempoMMSS(timerTiempoRestante);

        var box = document.getElementById('timer-box');
        if (box) box.classList.remove('timer-finished');
        actualizarBarraProgresoFondo();
    }

    cerrarPickerTiempo();
}

function formatearTiempoMMSS(segundosTotales) {
    var min = Math.floor(segundosTotales / 60);
    var seg = segundosTotales % 60;
    return (min < 10 ? "0" : "") + min + ":" + (seg < 10 ? "0" : "") + seg;
}

function toggleTemporizador() {
    if (timerTerminado) {
        detenerTemporizador();
        reiniciarTemporizador();
        return;
    }

    obtenerContextoAudio();

    var btn = document.getElementById('btn-start-timer');
    var box = document.getElementById('timer-box');

    if (timerEnEjecucion) {
        detenerTemporizador();
        if (btn) btn.innerText = idiomaActual === 'es' ? "Reanudar" : (idiomaActual === 'fr' ? "Reprendre" : "Resume");
    } else {
        if (timerTiempoRestante <= 0) return;
        timerEnEjecucion = true;
        if (btn) btn.innerText = idiomaActual === 'es' ? "Pausar" : (idiomaActual === 'fr' ? "Pause" : "Pause");
        if (box) box.classList.add('timer-running');
        
        timerInterval = setInterval(function() {
            if (timerTiempoRestante > 0) {
                timerTiempoRestante--;
                var display = document.getElementById('timer-display');
                if (display) display.innerText = formatearTiempoMMSS(timerTiempoRestante);
                actualizarBarraProgresoFondo();
            } else {
                detenerTemporizador();
                alertaTemporizadorCumplido();
            }
        }, 1000);
    }
}

    }
}

function alertaTemporizadorCumplido() {
    timerTerminado = true;
    var box = document.getElementById('timer-box');
    if (box) {
        box.classList.remove('timer-running');
        box.classList.add('timer-finished');
        box.style.backgroundImage = 'none';
    }
    var display = document.getElementById('timer-display');
    if (display) display.innerText = idiomaActual === 'es' ? "¡Tiempo cumplido! 🔔" : (idiomaActual === 'fr' ? "Temps écoulé ! 🔔" : "Time's up! 🔔");

    try {
        var ctx = obtenerContextoAudio();
        var ahora = ctx.currentTime;

        [0, 0.25, 0.5].forEach(function(offset) {
            var osc = ctx.createOscillator();
            var gain = ctx.createGain();

            osc.type = 'sine';
            osc.frequency.setValueAtTime(880, ahora + offset);
            osc.frequency.exponentialRampToValueAtTime(440, ahora + offset + 0.2);

            gain.gain.setValueAtTime(0.3, ahora + offset);
            gain.gain.exponentialRampToValueAtTime(0.01, ahora + offset + 0.2);

            osc.connect(gain);
            gain.connect(ctx.destination);

            osc.start(ahora + offset);
            osc.stop(ahora + offset + 0.2);
        });
    } catch(e) {}
}

function getProteinaKey() {

    if (protLC.includes('pescado')) return "-pescado";

    return "";
}

    var emojiIcon = (recetaElegida === "Sancocho Tradicional") ? "🍲" : (recetaElegida === "Empanadas Crujientes") ? "🥟" : (recetaElegida === "Coctel de Camarones") ? "🍤" : (recetaElegida === "Bandeja Paisa") ? "🍳" : (recetaElegida === "Arroz con Leche") ? "🍚" : (recetaElegida === "Ajiaco Santafereño") ? "🍲" : (recetaElegida === "Papa Rellena") ? "🥔" : (recetaElegida === "Postre Tres Leches") ? "🍰" : "🥥";

    if (cantidad === 0) {
        contenedor.innerHTML = "<span class='emoji-persona'>🍽️</span>";
    } else if (cantidad <= 10) {
        for (var i = 0; i < cantidad; i++) {
            var span = document.createElement('span');
            span.className = 'emoji-persona';
            span.innerText = emojiIcon;
            span.style.animationDelay = (i * 0.03) + "s";
            contenedor.appendChild(span);
        }
    } else {
        contenedor.innerHTML = '<span class="emoji-persona" style="font-size: 38px;">' + emojiIcon + '</span><span style="font-size: 28px; font-weight: 700; color: #2C422C; margin-left: 6px; animation: sombreritoSaltarin 0.5s ease-out;">x' + cantidad + '</span>';
    }
}

function validarPorciones() {
    if (porcionesElegidas === 0) {
        var msg = idiomaActual === 'es' ? "Por favor, selecciona para cuántas personas vas a cocinar hoy." : "Please select how many people you are cooking for today.";
        var title = idiomaActual === 'es' ? "Faltan Porciones" : "Missing Portions";
        mostrarAlerta(msg, title);
    } else {
        if (recetaElegida === "Cocadas Colombianas" || recetaElegida === "Coctel de Camarones" || recetaElegida === "Bandeja Paisa" || recetaElegida === "Arroz con Leche" || recetaElegida === "Ajiaco Santafereño" || recetaElegida === "Postre Tres Leches") {
            generarReceta();
        } else {
            irPantalla('screen-proteina');
        }
    }
}

function validarProteina() {
    if (proteinaElegida === "") {
        var msg = idiomaActual === 'es' ? "Por favor, selecciona una proteína para continuar." : "Please select a protein to continue.";
        var title = idiomaActual === 'es' ? "Falta Proteína" : "Missing Protein";
        mostrarAlerta(msg, title);
    } else {
        generarReceta();
    }
}

window.addEventListener("keydown", function(e) {
    if (document.getElementById('custom-modal').style.display === 'flex') {
        if (e.key === "Enter" || e.key === "Escape" || e.key === "Esc") { e.preventDefault(); cerrarAlerta(); }
        return;
    }

    if (document.getElementById('custom-upsell').style.display === 'flex') {
        if (e.key === "Enter") { e.preventDefault(); intentarUpgrade(); } 
        else if (e.key === "Escape" || e.key === "Esc") { e.preventDefault(); cerrarUpsell(); }
        return;
    }

    if (document.getElementById('custom-prompt').style.display === 'flex') {
        if (e.key === "Enter") { e.preventDefault(); aceptarPrompt(); } 
        else if (e.key === "Escape" || e.key === "Esc") { e.preventDefault(); cerrarPrompt(); }
        return;
    }

    if (document.getElementById('modal-timer-picker').style.display === 'flex') {
        if (e.key === "Enter") { e.preventDefault(); guardarTiempoPicker(); }
        else if (e.key === "Escape" || e.key === "Esc") { e.preventDefault(); cerrarPickerTiempo(); }
        return;
    }

    if (e.key === "Enter") {
        if (document.getElementById('screen-pin').classList.contains('active')) {
            e.preventDefault();
            verificarPIN();
            return;
        }
        var btnPrimary = document.querySelector(".screen.active .btn-primary");
        if (btnPrimary) { e.preventDefault(); btnPrimary.click(); }
    }

    if (e.key === "Escape" || e.key === "Esc") {
        var btnBack = document.querySelector(".screen.active .btn-back");
        if (btnBack) {
            e.preventDefault();
            btnBack.click();
        } else if (document.getElementById("screen-final").classList.contains("active")) {
            e.preventDefault();
            reiniciarApp();
        }
    }
});

document.addEventListener("click", function(e) {
    if (e.target.closest('.modal-overlay') || e.target.closest('.modal-card')) return;
    if (document.getElementById('custom-modal').style.display === 'flex' || document.getElementById('custom-prompt').style.display === 'flex' || document.getElementById('custom-upsell').style.display === 'flex' || document.getElementById('modal-timer-picker').style.display === 'flex') return;

    if (document.getElementById('screen-personas').classList.contains('active')) {
        if (!e.target.closest('.opcion-btn') && !e.target.closest('.btn-primary') && !e.target.closest('.btn-back') && !e.target.closest('.lang-container') && !e.target.closest('.lang-btn')) {
            porcionesElegidas = 0;
            document.querySelectorAll('.opcion-btn').forEach(function(btn) { btn.classList.remove('selected'); });
            document.getElementById('btn-mas').innerText = '+';
            actualizarPreviewPersonas(0);
        }
    }

    if (document.getElementById('screen-proteina').classList.contains('active')) {
        if (!e.target.closest('.proteina-card') && !e.target.closest('.btn-primary') && !e.target.closest('.btn-back')) {
            proteinaElegida = "";
            document.querySelectorAll('.proteina-card').forEach(function(card) { card.classList.remove('selected'); });
        }
    }
});
